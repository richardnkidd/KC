import React, { useState } from 'react';
import { Waves, MapPin, ChevronDown, AlertTriangle, ArrowUp, ArrowDown, Activity, Shield, Eye } from 'lucide-react';
import { useOceanConditions } from '../hooks/useOceanConditions';
import { cn } from '@/lib/utils';

// O'ahu beaches with safety monitoring
const OAHU_BEACHES = {
  "Ala Moana": { 
    lat: 21.2888, lng: -157.8394, 
    region: "Urban Core",
    hasLifeguards: true,
    shorebreakRisk: "low"
  },
  "Waikīkī": { 
    lat: 21.2793, lng: -157.8293, 
    region: "Town",
    hasLifeguards: true,
    shorebreakRisk: "low"
  },
  "Hanauma Bay": { 
    lat: 21.2693, lng: -157.6947, 
    region: "Southeast",
    hasLifeguards: true,
    shorebreakRisk: "moderate"
  },
  "Lanikai": { 
    lat: 21.3961, lng: -157.7275, 
    region: "Windward",
    hasLifeguards: false,
    shorebreakRisk: "low"
  },
  "Kailua": { 
    lat: 21.4022, lng: -157.7394, 
    region: "Windward",
    hasLifeguards: true,
    shorebreakRisk: "moderate"
  },
  "Pipeline": { 
    lat: 21.6650, lng: -158.0533, 
    region: "North Shore",
    hasLifeguards: true,
    shorebreakRisk: "extreme"
  },
  "Sunset Beach": { 
    lat: 21.6792, lng: -158.0417, 
    region: "North Shore",
    hasLifeguards: true,
    shorebreakRisk: "high"
  },
  "Makaha": { 
    lat: 21.4692, lng: -158.2208, 
    region: "Leeward Coast",
    hasLifeguards: true,
    shorebreakRisk: "high"
  },
  "Sandy Beach": { 
    lat: 21.2853, lng: -157.6722, 
    region: "Southeast",
    hasLifeguards: true,
    shorebreakRisk: "extreme"
  },
  "Makapuu": { 
    lat: 21.3097, lng: -157.6494, 
    region: "Southeast",
    hasLifeguards: false,
    shorebreakRisk: "high"
  }
} as const;

export const OceanConditionsCard: React.FC = () => {
  const [selectedBeach, setSelectedBeach] = useState('Ala Moana');
  const [isDropdownOpen, setIsDropdownOpen] = useState(false);
  const [activeTab, setActiveTab] = useState<'conditions' | 'safety'>('conditions');
  
  const { data: oceanData, isLoading, error } = useOceanConditions(selectedBeach);

  if (error) {
    return (
      <div className="text-center py-8">
        <div className="w-16 h-16 rounded-2xl bg-[#8B2323]/10 flex items-center justify-center mx-auto mb-4">
          <Waves className="w-8 h-8 text-[#8B2323]" />
        </div>
        <p className="text-base font-medium text-primary">Ocean Conditions Unavailable</p>
        <p className="text-sm text-muted mt-1">Unable to fetch surf and safety data</p>
      </div>
    );
  }

  if (isLoading || !oceanData) {
    return (
      <div className="animate-pulse">
        <div className="flex items-center space-x-4 mb-6">
          <div className="w-12 h-12 bg-[#2D2D2B]/30 rounded-2xl"></div>
          <div className="space-y-2">
            <div className="h-5 bg-[#2D2D2B]/30 rounded-xl w-32"></div>
            <div className="h-4 bg-[#2D2D2B]/20 rounded-xl w-24"></div>
          </div>
        </div>
        <div className="space-y-3">
          <div className="h-20 bg-[#2D2D2B]/30 rounded-2xl"></div>
          <div className="h-16 bg-[#2D2D2B]/30 rounded-2xl"></div>
        </div>
      </div>
    );
  }

  const beachInfo = OAHU_BEACHES[selectedBeach as keyof typeof OAHU_BEACHES];
  
  const getShorebreakColor = (risk: string) => {
    switch (risk) {
      case 'low': return 'text-[#10B981] bg-[#10B981]/20';
      case 'moderate': return 'text-[#F59E0B] bg-[#F59E0B]/20';
      case 'high': return 'text-[#EF4444] bg-[#EF4444]/20';
      case 'extreme': return 'text-[#8B2323] bg-[#8B2323]/20';
      default: return 'text-[#6B7280] bg-[#6B7280]/20';
    }
  };

  const getBacteriaLevel = (level: string) => {
    switch (level) {
      case 'safe': return { text: 'Safe', color: 'text-[#10B981]', bg: 'bg-[#10B981]/20' };
      case 'caution': return { text: 'Caution', color: 'text-[#F59E0B]', bg: 'bg-[#F59E0B]/20' };
      case 'warning': return { text: 'High Count', color: 'text-[#EF4444]', bg: 'bg-[#EF4444]/20' };
      default: return { text: 'Unknown', color: 'text-[#6B7280]', bg: 'bg-[#6B7280]/20' };
    }
  };

  const getJellyfishRisk = () => {
    const daysAfterFullMoon = oceanData.beachSafety.jellyfishAlert.daysAfterFullMoon;
    if (daysAfterFullMoon >= 8 && daysAfterFullMoon <= 10) {
      return { text: 'High Risk', color: 'text-[#EF4444]', bg: 'bg-[#EF4444]/20', icon: '🎐' };
    } else if (daysAfterFullMoon >= 6 && daysAfterFullMoon <= 12) {
      return { text: 'Moderate Risk', color: 'text-[#F59E0B]', bg: 'bg-[#F59E0B]/20', icon: '⚠️' };
    }
    return { text: 'Low Risk', color: 'text-[#10B981]', bg: 'bg-[#10B981]/20', icon: '✅' };
  };

  const bacteriaStatus = getBacteriaLevel(oceanData.beachSafety.bacteriaLevel);
  const jellyfishRisk = getJellyfishRisk();

  return (
    <div>
      {/* Header */}
      <div className="flex items-center justify-between mb-6">
        <div className="flex items-center space-x-4">
          <div className="relative animate-float">
            <div className="w-12 h-12 rounded-2xl flex items-center justify-center overflow-hidden">
              <div className="absolute inset-0 bg-gradient-to-br from-[#20B2AA] to-[#008B8B]"></div>
              <Waves className="w-6 h-6 text-white relative z-10" />
            </div>
          </div>
          <div className="flex-1">
            <h2 className="text-h2 font-display text-emphasis-high">Ocean Conditions</h2>
            
            {/* Beach Selector */}
            <div className="relative mt-1">
              <button
                onClick={() => setIsDropdownOpen(!isDropdownOpen)}
                className="flex items-center gap-2 text-small text-emphasis-medium hover:text-emphasis-high transition-colors group"
              >
                <MapPin className="w-3 h-3" />
                <span>{selectedBeach}</span>
                <ChevronDown className={cn(
                  "w-3 h-3 transition-transform",
                  isDropdownOpen && "rotate-180"
                )} />
              </button>

              {/* Beach Dropdown */}
              {isDropdownOpen && (
                <>
                  <div 
                    className="fixed inset-0 z-40" 
                    onClick={() => setIsDropdownOpen(false)}
                  />
                  <div className="absolute top-full mt-2 w-72 bg-white rounded-2xl shadow-lg border border-[rgba(0,0,0,0.06)] overflow-hidden z-50">
                    <div className="max-h-64 overflow-y-auto">
                      {Object.entries(
                        Object.entries(OAHU_BEACHES).reduce((acc, [name, info]) => {
                          if (!acc[info.region]) acc[info.region] = [];
                          acc[info.region].push(name);
                          return acc;
                        }, {} as Record<string, string[]>)
                      ).map(([region, beaches]) => (
                        <div key={region}>
                          <div className="px-4 py-2 text-xs font-medium text-[#333333]/50 bg-[#F5FAF8]">
                            {region}
                          </div>
                          {beaches.map((beach) => (
                            <button
                              key={beach}
                              onClick={() => {
                                setSelectedBeach(beach);
                                setIsDropdownOpen(false);
                              }}
                              className={cn(
                                "w-full text-left px-4 py-2 text-sm hover:bg-[#F5FAF8] transition-colors flex items-center justify-between",
                                selectedBeach === beach && "bg-[#EEE0C9]/30 text-[#214263] font-medium"
                              )}
                            >
                              <span>{beach}</span>
                              <div className="flex items-center gap-1">
                                {OAHU_BEACHES[beach as keyof typeof OAHU_BEACHES].hasLifeguards && (
                                  <Shield className="w-3 h-3 text-[#10B981]" />
                                )}
                                <div className={cn(
                                  "w-2 h-2 rounded-full",
                                  getShorebreakColor(OAHU_BEACHES[beach as keyof typeof OAHU_BEACHES].shorebreakRisk).replace('text-', 'bg-').replace('/20', '')
                                )}></div>
                              </div>
                            </button>
                          ))}
                        </div>
                      ))}
                    </div>
                  </div>
                </>
              )}
            </div>
          </div>
        </div>

        <div className="text-right">
          <p className="text-caption text-emphasis-low">Wave Height</p>
          <p className="text-body font-semibold gradient-text">{oceanData.surf.waveHeight}ft</p>
        </div>
      </div>

      {/* Tab Navigation */}
      <div className="flex mb-6">
        <button
          onClick={() => setActiveTab('conditions')}
          className={cn(
            "flex-1 px-4 py-2 text-sm font-medium rounded-l-xl transition-colors",
            activeTab === 'conditions' 
              ? "bg-[#407B9E] text-white" 
              : "bg-[#407B9E]/10 text-[#407B9E] hover:bg-[#407B9E]/20"
          )}
        >
          🌊 Surf & Tides
        </button>
        <button
          onClick={() => setActiveTab('safety')}
          className={cn(
            "flex-1 px-4 py-2 text-sm font-medium rounded-r-xl transition-colors",
            activeTab === 'safety' 
              ? "bg-[#407B9E] text-white" 
              : "bg-[#407B9E]/10 text-[#407B9E] hover:bg-[#407B9E]/20"
          )}
        >
          🏖️ Beach Safety
        </button>
      </div>

      {/* Tab Content */}
      {activeTab === 'conditions' ? (
        <div className="space-y-4">
          {/* Current Surf Conditions */}
          <div className="glass rounded-2xl p-4">
            <div className="flex items-center justify-between mb-4">
              <h3 className="text-sm font-medium text-secondary">Current Surf</h3>
              <span className={cn(
                "px-3 py-1 rounded-full text-xs font-medium",
                oceanData.surf.waveHeight < 2 ? "bg-[#6B7280]/20 text-[#6B7280]" :
                oceanData.surf.waveHeight < 4 ? "bg-[#3B82F6]/20 text-[#3B82F6]" :
                oceanData.surf.waveHeight < 6 ? "bg-[#F59E0B]/20 text-[#F59E0B]" :
                "bg-[#EF4444]/20 text-[#EF4444]"
              )}>
                {oceanData.surf.waveHeight < 2 ? 'Small' : 
                 oceanData.surf.waveHeight < 4 ? 'Fun Size' :
                 oceanData.surf.waveHeight < 6 ? 'Solid' : 'Firing! 🔥'}
              </span>
            </div>
            
            <div className="grid grid-cols-3 gap-4 text-center">
              <div>
                <p className="text-caption text-emphasis-low mb-1">Swell</p>
                <p className="text-lg font-medium text-primary">{oceanData.surf.swellDirection}° / {oceanData.surf.swellPeriod}s</p>
              </div>
              <div>
                <p className="text-caption text-emphasis-low mb-1">Wind</p>
                <p className="text-lg font-medium text-primary">{oceanData.surf.windSpeed}kn {oceanData.surf.windDirection}</p>
              </div>
              <div>
                <p className="text-caption text-emphasis-low mb-1">Quality</p>
                <p className="text-lg font-medium text-primary">{oceanData.surf.surfQuality}/10</p>
              </div>
            </div>
          </div>

          {/* Tide Information */}
          <div className="glass rounded-2xl p-4">
            <div className="flex items-center justify-between mb-4">
              <h3 className="text-sm font-medium text-secondary">Tide Status</h3>
              <span className="text-body font-semibold text-emphasis-high">{oceanData.tides.current.status}</span>
            </div>

            {/* Tide Level Visualization */}
            <div className="mb-4">
              <div className="flex items-center justify-between mb-2">
                <span className="text-small text-emphasis-medium">Current Level</span>
                <span className="text-body font-semibold text-emphasis-high">{oceanData.tides.current.level} ft</span>
              </div>

              <div className="relative">
                <div className="w-full rounded-full overflow-hidden h-3 bg-[rgb(var(--tropical-stone)/0.3)]">
                  <div 
                    className="h-full rounded-full transition-all duration-1000 ease-out bg-gradient-to-r from-[rgb(var(--tropical-ocean))] to-[rgb(var(--tropical-ocean-light))]"
                    style={{ width: `${oceanData.tides.current.percentage}%` }}
                  >
                    <div className="absolute right-0 top-1/2 -translate-y-1/2 w-3 h-3 bg-white rounded-full shadow-lg"></div>
                  </div>
                </div>
                <div className="flex justify-between text-xs text-muted mt-1">
                  <span>Low</span>
                  <span>High</span>
                </div>
              </div>
            </div>

            {/* Next Tides */}
            <div className="space-y-2">
              {oceanData.tides.upcoming.slice(0, 2).map((tide, index) => (
                <div key={index} className="flex items-center justify-between p-2 rounded-xl bg-[rgb(var(--tropical-stone)/0.05)]">
                  <div className="flex items-center space-x-2">
                    {tide.type === 'High' ? (
                      <ArrowUp className="w-4 h-4 text-[#407B9E]" />
                    ) : (
                      <ArrowDown className="w-4 h-4 text-[#EEE0C9]" />
                    )}
                    <span className="text-sm font-medium text-primary">{tide.type} Tide</span>
                  </div>
                  <div className="text-right">
                    <p className="text-sm font-medium text-primary">{tide.time}</p>
                    <p className="text-xs text-muted">{tide.height} ft</p>
                  </div>
                </div>
              ))}
            </div>
          </div>
        </div>
      ) : (
        <div className="space-y-4">
          {/* Beach Safety Status */}
          <div className="glass rounded-2xl p-4">
            <div className="flex items-center justify-between mb-4">
              <h3 className="text-sm font-medium text-secondary">Safety Overview</h3>
              <div className="flex items-center gap-2">
                {beachInfo.hasLifeguards && (
                  <div className="flex items-center gap-1 px-2 py-1 rounded-full bg-[#10B981]/20">
                    <Shield className="w-3 h-3 text-[#10B981]" />
                    <span className="text-xs text-[#10B981] font-medium">Lifeguards</span>
                  </div>
                )}
              </div>
            </div>

            <div className="grid grid-cols-2 gap-3">
              <div className={cn("p-3 rounded-xl", getShorebreakColor(beachInfo.shorebreakRisk))}>
                <p className="text-xs font-medium mb-1">Shorebreak Risk</p>
                <p className="text-sm font-bold capitalize">{beachInfo.shorebreakRisk}</p>
              </div>
              <div className={cn("p-3 rounded-xl", bacteriaStatus.bg)}>
                <p className="text-xs font-medium mb-1">Water Quality</p>
                <p className={cn("text-sm font-bold", bacteriaStatus.color)}>{bacteriaStatus.text}</p>
              </div>
            </div>
          </div>

          {/* Jellyfish Alert */}
          <div className={cn("glass rounded-2xl p-4 border-l-4", 
            jellyfishRisk.color.includes('EF4444') ? 'border-[#EF4444]' :
            jellyfishRisk.color.includes('F59E0B') ? 'border-[#F59E0B]' : 'border-[#10B981]'
          )}>
            <div className="flex items-center justify-between mb-3">
              <h3 className="text-sm font-medium text-secondary flex items-center gap-2">
                {jellyfishRisk.icon} Box Jellyfish Alert
              </h3>
              <span className={cn("px-3 py-1 rounded-full text-xs font-medium", jellyfishRisk.bg, jellyfishRisk.color)}>
                {jellyfishRisk.text}
              </span>
            </div>
            <p className="text-xs text-muted mb-2">
              {oceanData.beachSafety.jellyfishAlert.daysAfterFullMoon} days after full moon
            </p>
            <p className="text-sm text-primary">
              {oceanData.beachSafety.jellyfishAlert.prediction}
            </p>
          </div>

          {/* Current Warnings */}
          {oceanData.beachSafety.currentWarnings.length > 0 && (
            <div className="glass rounded-2xl p-4">
              <h3 className="text-sm font-medium text-secondary mb-3 flex items-center">
                <AlertTriangle className="w-4 h-4 mr-2 text-[#F59E0B]" />
                Active Warnings
              </h3>
              {oceanData.beachSafety.currentWarnings.map((warning, index) => (
                <div key={index} className="flex items-start space-x-3 mb-3 last:mb-0">
                  <div className="w-8 h-8 rounded-lg bg-[#F59E0B]/20 flex items-center justify-center">
                    <AlertTriangle className="w-4 h-4 text-[#F59E0B]" />
                  </div>
                  <div className="flex-1">
                    <p className="text-sm font-medium text-primary">{warning.type}</p>
                    <p className="text-xs text-muted mt-1">{warning.description}</p>
                  </div>
                </div>
              ))}
            </div>
          )}

          {/* Wildlife Sightings */}
          {oceanData.beachSafety.wildlifeSightings.length > 0 && (
            <div className="glass rounded-2xl p-4">
              <h3 className="text-sm font-medium text-secondary mb-3 flex items-center">
                <Eye className="w-4 h-4 mr-2 text-[#10B981]" />
                Recent Sightings
              </h3>
              {oceanData.beachSafety.wildlifeSightings.map((sighting, index) => (
                <div key={index} className="flex items-center justify-between p-2 rounded-xl bg-[#10B981]/10 mb-2 last:mb-0">
                  <div className="flex items-center space-x-2">
                    <span className="text-lg">{sighting.animal === 'monk-seal' ? '🦭' : '🐢'}</span>
                    <div>
                      <p className="text-sm font-medium text-primary">
                        {sighting.animal === 'monk-seal' ? 'Hawaiian Monk Seal' : 'Hawaiian Green Sea Turtle'}
                      </p>
                      <p className="text-xs text-muted">{sighting.location}</p>
                    </div>
                  </div>
                  <p className="text-xs text-muted">{sighting.timeAgo}</p>
                </div>
              ))}
            </div>
          )}
        </div>
      )}

      {/* Footer */}
      <div className="mt-6 pt-4 border-t border-[rgb(var(--tropical-sage)/0.2)]">
        <p className="text-xs text-muted flex items-center">
          <Activity className="w-3 h-3 mr-2" />
          NOAA • Ocean Safety • DLNR • Updates every 15 minutes
        </p>
      </div>
    </div>
  );
};
